let wind = false;
let goneBall = []; // goneBall is an array.
let stayBall;
let speedInitialBoth = 0.03;
let xMinBoth = -10;
let xMaxBoth = 10;
let yMinBoth = -10;
let yMaxBoth = 10;

// Constructor function for the stay ball, essentially object.
function StayBall(x, y, d) {
  this.x = x;
  this.y = y;
  this.d = d;
  this.speedInitial = speedInitialBoth;
  
  this.display = function() {
    fill("#92c262");
    noStroke();
    circle(this.x, this.y, this.d);
  };
  this.move = function() {
    this.x += random(xMinBoth, xMaxBoth) * this.speedInitial;
    this.y += random(yMinBoth, yMaxBoth) * this.speedInitial;
  };
}


// Constructor function for the gone balls, essentially object.
function GoneBall(x, y, d) {
  this.x = x;
  this.y = y;
  this.d = d;
  this.angle = random(0, 90);
  this.baseSpeed = random(1, 3);
  this.speedInitial = speedInitialBoth;
  this.t = 0;
  
  this.display = function() {
    fill("#92c262");
    noStroke();
    circle(this.x, this.y, this.d);
  };
  this.move = function() {
    if(wind == true) {
      this.t ++;
      let currentSpeed = this.baseSpeed * (this.t * 0.01);
      this.x += currentSpeed * cos(this.angle) + 1.5;
      this.y -= currentSpeed * abs(sin(this.angle)) + 0.5;
    } else {
      this.x += random(xMinBoth, xMaxBoth) * this.speedInitial;
      this.y += random(yMinBoth, yMaxBoth) * this.speedInitial;
    }
  }; 
}


function setup() {
  createElement('p', 'try to click the canvas with your mouse.');
  angleMode(DEGREES) // equals to createCanvas, is p5 internal API, can only be used inside setup/draw, non global.
  createCanvas(400, 400);
  
  for(let i = 0; i < 180; i++) {
    goneBall.push(new GoneBall(random(0, width), random(0, height), 50));
  } 
  // goneBall array inherits GoneBall.
  stayBall = new StayBall(width / 3, height * 2 /3, 50);
  // stayBall inherits StayBall.
} 

function draw() {
  background(240);
  
  
  for (let i = 0; i < goneBall.length; i++) {
    goneBall[i].move();
    goneBall[i].display();
  }
  
  stayBall.display();
  stayBall.move();
}

function mousePressed() {
    wind = true;
}
