Whether the Button is Pressed?

Sometimes there is pinMode in the code, and sometimes there isn’t. Why is that? Right now I’m insisting on adding pinMode in every sketch in order to keep my understanding consistent.

int lastButtonState = LOW;
int inputPin = 2;


void setup() {
  Serial.begin(9600);
  pinMode(inputPin, INPUT);

}

void loop() {
  int buttonState = digitalRead(inputPin);
  if (buttonState != lastButtonState) {
    if (buttonState == HIGH) {
      Serial.println("Button was just pressed.");
    }
  }

  lastButtonState = buttonState;

Figure it out in the class: Analog input pins don't need a pinMode. Analog output pins don't hurt with a pinMode. Digital (input/output) pins do need one cuz they can do both input and output so we need to define. But just keep defining pinMode for every pin.

How Long is the Button Pressed?
int lastButtonState = LOW;
int inputPin = 2;

int buttonPresses = 0;

long pressTime = 0;


void setup() {
  Serial.begin(9600);
  pinMode(inputPin, INPUT);

}

void loop() {
  int buttonState = digitalRead(inputPin);
  if (buttonState != lastButtonState) {
    if (buttonState == HIGH) {
      buttonPresses++;
      Serial.println("Button has been pressed");
      Serial.println(buttonPresses);
      Serial.println("times.");

      pressTime = millis();
    }

    if (buttonState == LOW) {
      long holdTime = millis() - pressTime;
      Serial.println("Button has been pressed for");
      Serial.println(holdTime);
      Serial.println("milliseconds.");
    }
  }

  lastButtonState = buttonState;
}
State Change Detection on Analog Sensors
Threshold Crossing
int inputPin = A0;

int lastSensorState = LOW;
int threshold = 512; //an arbitrary threshold value


void setup() {
  Serial.begin(9600);
  pinMode(inputPin, INPUT);

}

void loop() {
  int sensorState = analogRead(inputPin);

  if (sensorState >= threshold && lastSensorState < threshold) {
    Serial.println("Sensor crossed the threshold.");
  }

  lastSensorState = sensorState;
}
Peak Detection
int inputPin = A0;

int peakValue = 0;
int threshold = 50;
int noise = 5; //value depends on different sensors.


void setup() {
  Serial.begin(9600);
  pinMode(inputPin, INPUT);

}

void loop() {
  int sensorValue = analogRead(inputPin);

  if (sensorValue > peakValue) {
    peakValue = sensorValue; 
  }
  if (sensorValue <= threshold - noise && peakValue > threshold + noise) {
    Serial.println(peakValue);
    peakValue = 0;
  }
}

  Wire.begin();

void loop() {
  // ==== 读取 TMP102 温度 ====
  Wire.beginTransmission(0x48); // TMP102 I2C 地址
  Wire.write(0x00);             // 温度寄存器
  Wire.endTransmission();
  Wire.requestFrom(0x48, 2);    // 读取两个字节
  
  float temperatureC = 0;
  if (Wire.available() >= 2) {
    int raw = Wire.read() << 8 | Wire.read();
    raw >>= 4; // TMP102 是 12 位数据
    temperatureC = raw * 0.0625; // 每 LSB = 0.0625 °C
  }
  
  Serial.print("Temp = ");
  Serial.print(temperatureC);
  Serial.println(" C");

  // ==== 温度映射到 LED 亮度 ====
  // 假设 20°C → 0（灭），40°C → 255（最亮）
  int ledBrightness = map(temperatureC, 20, 40, 0, 255);

  // 限制范围，避免小于 20 或大于 40 的情况出错
  ledBrightness = constrain(ledBrightness, 0, 255);

  // PWM 输出控制 LED 亮度
  analogWrite(ledPin, ledBrightness);

  delay(500); // 半秒更新一次
}
