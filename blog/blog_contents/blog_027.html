Lab 01-SPI Lab-Data Logging With an SD Card Reader using SPI Communication

Connect a DHT-11 Temp & Humidity Sensor to a Microcontroller
Connect a SD Card Reader and Store the Data Read in the SD Card
#include 
#include 
#include "DHT.h"

// SD 卡设置
const int chipSelect = 10;
const char fileName[] = "datalog.csv";
long lastReading = 0;

// DHT 传感器设置
#define DHTPIN 2        // DHT11 数据引脚接 D2
#define DHTTYPE DHT11   // DHT11 类型（如果是 DHT22 就改成 DHT22）
DHT dht(DHTPIN, DHTTYPE);  

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);
  Serial.begin(9600);
  Serial.println("Initializing SD card...");

  // 初始化 SD 卡
  if (!SD.begin(chipSelect)) {
    Serial.println("SD card initialization failed!");
    while (true) {
      digitalWrite(LED_BUILTIN, HIGH);
    }
  }

  Serial.println("SD card initialized successfully.");
  dht.begin(); // 初始化温湿度传感器
  digitalWrite(LED_BUILTIN, LOW);
}

void loop() {
  // 每 2 秒记录一次温湿度
  if (millis() - lastReading > 2000) {
    float humidity = dht.readHumidity();
    float temperature = dht.readTemperature(); // 摄氏度

    // 检查读取是否成功
    if (isnan(humidity) || isnan(temperature)) {
      Serial.println("Failed to read from DHT sensor!");
      return;
    }

    // 拼接成 CSV 字符串：时间,温度,湿度
    String dataString = "";
    dataString += String(millis() / 1000.0, 1); // 秒为单位的时间戳
    dataString += ",";
    dataString += String(temperature, 1);
    dataString += ",";
    dataString += String(humidity, 1);

    // 打开 SD 文件（追加写入）
    File dataFile = SD.open(fileName, FILE_WRITE);

    if (dataFile) {
      digitalWrite(LED_BUILTIN, HIGH);
      dataFile.println(dataString);
      dataFile.close();
      digitalWrite(LED_BUILTIN, LOW);
      lastReading = millis();

      // 同时在串口打印
      Serial.print("Logged: ");
      Serial.println(dataString);
    } else {
      Serial.println("Error opening datalog.csv");
      digitalWrite(LED_BUILTIN, HIGH);
    }
  }
}

Lab 02-I2C Lab-OLED Screen Display using I2C

This week, I was trying to make a little birthday gift for my friend in China. Basically, this gift involves an OLED screen showing "Happy Birthday Hazel" in Chinese and a speaker playing Birthday Song tone. Therefore, I believe that the process of making this gift can substitue this lab.

Remember to download the "u8g2" by oliver library.

#include 
#include 
#include 

#define ROTATION U8G2_R0

U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(ROTATION, /* reset=*/ U8X8_PIN_NONE);

const char* text = "榛子生日快乐";

// speaker pin
#define SPEAKER_PIN 2

// 4 Leds pin
int ledPins[4] = {3, 5, 7, 9};

// Birthday Song notes (frequency sheet)
int melody[] = {
  262, 262, 294, 262, 349, 330, // Happy Birthday to you
  262, 262, 294, 262, 392, 349, // Happy Birthday to you
  262, 262, 523, 440, 349, 330, 294, // Happy Birthday to you
  466, 466, 440, 349, 392, 349 // Happy Birthday to you
};

// note time
// 2 = half note = 2 beats = 500ms
// 4 = quarter note = 1 beat = 250ms
// 8 = eighth note = 1/2 beat = 125ms
int noteDurations[] = {
  4, 4, 4, 4, 4, 2,
  4, 4, 4, 4, 4, 2,
  4, 4, 4, 4, 4, 4, 2,
  4, 4, 4, 4, 4, 2
};

int length = sizeof(melody) / sizeof(melody[0]);

// draw contents on the screen
void drawScreen(int16_t centerY) {
  u8g2.clearBuffer();

  // texts
  int16_t textW = u8g2.getUTF8Width(text);
  int16_t x = (128 - textW) / 2;
  int16_t y = centerY - 17;
  u8g2.drawUTF8(x, y, text);

  // graphic
  int16_t centerX = 64;
  u8g2.drawEllipse(centerX, centerY, 8, 10, U8G2_DRAW_ALL);
  u8g2.drawTriangle(centerX - 9, centerY - 5, centerX + 9, centerY - 5, centerX, centerY - 12);
  u8g2.drawLine(centerX, centerY - 2, centerX, centerY + 3);
  u8g2.drawLine(centerX - 3, centerY - 1, centerX - 3, centerY + 2);
  u8g2.drawLine(centerX + 3, centerY - 1, centerX + 3, centerY + 2);

  u8g2.sendBuffer();
}

void setup() {
  pinMode(SPEAKER_PIN, OUTPUT);
  for (int i = 0; i < 4; i++) {
    pinMode(ledPins[i], OUTPUT);
    digitalWrite(ledPins[i], LOW);
  }

  u8g2.begin();
  u8g2.setFont(u8g2_font_wqy12_t_gb2312);
  u8g2.setFontMode(1);
}

void loop() {
  int16_t baseY = 45; // initial Y
  int16_t jumpY = baseY - 5; // jump Y

  for (int i = 0; i < length; i++) {
    // Led control
    int ledIndex = i % 4; // 0~3 loop
    for (int j = 0; j < 5; j++) {
      digitalWrite(ledPins[j], (j == ledIndex) ? HIGH : LOW);
    }

    // jumping
    int16_t currentY = (i % 2 == 0) ? jumpY : baseY;
    drawScreen(currentY);

    // play notes in speaker
    int noteDuration = 1000 / noteDurations[i];
    tone(SPEAKER_PIN, melody[i], noteDuration);
    delay(noteDuration * 1.30);
    noTone(SPEAKER_PIN);
  }

  // stops for 2 seconds and loop again
  for (int j = 0; j < 5; j++) digitalWrite(ledPins[j], LOW);
  delay(2000);
}
