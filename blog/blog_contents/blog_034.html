What I Built for my Prototype

I used P5.js to create a random-edge effect for the images.

let imgs = [];
let numImages = 5;
let currentIndex = 0;
let jaggedAmount = 7;
let jaggedFrequency = 0.05;

function preload() {
  for (let i = 0; i < numImages; i++) {
    imgs[i] = loadImage(`images/${i}.jpg`);
  }
}

function setup() {
  createCanvas(800, 800);
  imageMode(CENTER);
  noLoop();

  processNextImage();
}

function processNextImage() {
  if (currentIndex >= imgs.length) {
    console.log("All compleated.");
    return;
  }

  let img = imgs[currentIndex];
  console.log("Handling img:", currentIndex);

  drawJaggedImage(img, currentIndex);

  currentIndex++;
  setTimeout(processNextImage, 300);
}

function drawJaggedImage(img, index) {
  clear();

  let cx = width / 2;
  let cy = height / 2;
  let w = img.width;
  let h = img.height;

  image(img, cx, cy);
  img.loadPixels();

  strokeWeight(1);

  drawJaggedEdge(cx, cy, w, h, img, "top");
  drawJaggedEdge(cx, cy, w, h, img, "bottom");
  drawJaggedEdge(cx, cy, w, h, img, "left");
  drawJaggedEdge(cx, cy, w, h, img, "right");

  saveCanvas(`jagged_${index}`, "png");
}

function drawJaggedEdge(cx, cy, w, h, img, side) {
  for (let i = 0; i < (side === "top" || side === "bottom" ? w : h); i++) {
    let n = noise(i * jaggedFrequency);
    let offset = map(n, 0, 1, -jaggedAmount, jaggedAmount);

    let imgX = 0,
      imgY = 0;
    if (side === "top") {
      imgX = i;
      imgY = 0;
    }
    if (side === "bottom") {
      imgX = i;
      imgY = h - 1;
    }
    if (side === "left") {
      imgX = 0;
      imgY = i;
    }
    if (side === "right") {
      imgX = w - 1;
      imgY = i;
    }

    let idx = (imgX + imgY * w) * 4;
    let r = img.pixels[idx];
    let g = img.pixels[idx + 1];
    let b = img.pixels[idx + 2];

    if (offset >= 0) {
      stroke(r, g, b);
      noErase();
    } else {
      erase();
    }

    if (side === "top") {
      let x = cx - w / 2 + i;
      let y = cy - h / 2;
      line(x, y, x, y - offset);
    }

    if (side === "bottom") {
      let x = cx - w / 2 + i;
      let y = cy + h / 2;
      line(x, y, x, y + offset);
    }

    if (side === "left") {
      let x = cx - w / 2;
      let y = cy - h / 2 + i;
      line(x, y, x - offset, y);
    }

    if (side === "right") {
      let x = cx + w / 2;
      let y = cy - h / 2 + i;
      line(x, y, x + offset, y);
    }

    noErase();
  }
}
