2-Output: Watering Pump

In testing, it was observed that after the water pump operated normally once, its subsequent operation lasted significantly longer than the duration specified in the code. I assumed this may be caused by the pump motor exerting a reverse effect on the circuit. I added a pull-down resistor and a capacitor in the circuit but failed in resolving that problem.

After the office hour with Tom, I made the following changes:

1. Change the NPN TIP 102 to MOSFET.

2. Move all the components to the right side of the breadboard to make them closer to the 3.3V pin.

3. Change to a bigger capacitor.

4. Replace with a bigger diode.

5. Get a new pump.

Done all these above, the problem is somehow solved. I guess this is the physical world. What will run well logically will not act the same physically. Sometimes I just need to change some components. And specification of the components matter very much.

***When next time working with high-current motors, remember: motors will cost a suddent high currency comsumption when starting, which can easily eat the currency that Arduino needs to function. (Because they are in the same GND.) Therefore, sufficient power supply, diode, capacitor, pull-down resisitor are demanded.

const char* mqttServer = "34.69.76.141";
const int   mqttPort   = 1883;

const char* TOPIC_MOISTURE = "plant/moisture";
const char* TOPIC_WATER    = "plant/water";

//const unsigned long PUBLISH_INTERVAL_MS = 5 * 60 * 60 * 1000UL; //5 hours
const unsigned long PUBLISH_INTERVAL_MS = 5000UL; //5 seconds

const int PUMP_PIN = 9;
const unsigned long DEFAULT_PUMP_MS = 1500;
const unsigned long PUMP_MAX_MS = 2000;

const unsigned long SENSOR_SETTLE_MS = 2000;

bool pumpOn = false;
unsigned long pumpOffAtMs = 0;
unsigned long pumpStartedAtMs = 0;
unsigned long pumpStoppedAtMs = 0;

WiFiClient wifiClient;
PubSubClient client(wifiClient);
SparkFunSoilMoistureSensor soil;

unsigned long lastPublishMs = 0;

void pumpStartFor(unsigned long durationMs) {
  if (durationMs < 50) durationMs = 50;
  if (durationMs > PUMP_MAX_MS) durationMs = PUMP_MAX_MS;

  digitalWrite(PUMP_PIN, HIGH);
  pumpOn = true;
  pumpStartedAtMs = millis();
  pumpOffAtMs = pumpStartedAtMs + durationMs;

  Serial.print("[PUMP] ON for ");
  Serial.print(durationMs);
  Serial.println(" ms");
}

void pumpStopNow() {
  digitalWrite(PUMP_PIN, LOW);
  pumpOn = false;
  pumpOffAtMs = 0;
  pumpStartedAtMs = 0;
  pumpStoppedAtMs = millis();
  Serial.println("[PUMP] OFF");
}

int readMoistureStable() {
  long sum = 0;
  for (int i = 0; i < 5; i++) {
    sum += soil.readMoistureValue();
    delay(20);
  }
  int raw = (int)(sum / 5);
  if (raw < 0) raw = 0;
  if (raw > 1023) raw = 1023;
  return 1023 - raw;
}

void onMqttMessage(char* topic, byte* payload, unsigned int length) {
  Serial.print("[MQTT] ");
  Serial.print(topic);
  Serial.print(" -> ");

  String msg;
  msg.reserve(length);
  for (unsigned int i = 0; i < length; i++) msg += (char)payload[i];
  msg.trim();
  Serial.println(msg);

  if (String(topic) != TOPIC_WATER) return;

  if (msg.equalsIgnoreCase("off") || msg.equalsIgnoreCase("0")) {
    pumpStopNow();
    return;
  }

  if (msg.equalsIgnoreCase("on")) {
    pumpStartFor(DEFAULT_PUMP_MS);
    return;
  }

  if (msg.startsWith("on ")) {
    String t = msg.substring(3);
    t.trim();
    unsigned long ms = (unsigned long)t.toInt();
    if (ms > 0) pumpStartFor(ms);
    else pumpStartFor(DEFAULT_PUMP_MS);
    return;
  }

  long ms = msg.toInt();
  if (ms > 0) {
    pumpStartFor((unsigned long)ms);
    return;
  }

  Serial.println("[PUMP] Unknown command");
}

void connectWifiOnce() {
  if (WiFi.status() == WL_CONNECTED) return;

  Serial.print("[WiFi] Connecting ");
  Serial.println(ssid);

  WiFi.begin(ssid, pass);

  unsigned long t0 = millis();
  while (WiFi.status() != WL_CONNECTED && (millis() - t0) < 15000UL) {
    delay(250);
    Serial.print(".");
  }
  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    Serial.print("[WiFi] OK, IP=");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("[WiFi] Timeout, will retry");
  }
}

void connectMqttOnce() {
  if (client.connected()) return;
  if (WiFi.status() != WL_CONNECTED) return;

  client.setServer(mqttServer, mqttPort);
  client.setCallback(onMqttMessage);

  String clientId = "nano33iot-" + String((uint32_t)random(0xFFFFFF), HEX);

  Serial.print("[MQTT] Connecting... ");
  if (client.connect(clientId.c_str())) {
    Serial.println("OK");
    client.subscribe(TOPIC_WATER);
    Serial.println("[MQTT] Subscribed plant/water");
  } else {
    Serial.print("FAIL state=");
    Serial.println(client.state());
  }
}

void publishMoistureNow() {
  unsigned long now = millis();
  if (pumpOn) return;
  if (pumpStoppedAtMs && (now - pumpStoppedAtMs) < SENSOR_SETTLE_MS) return;

  int moisture = readMoistureStable();

  char msg[16];
  itoa(moisture, msg, 10);

  client.publish(TOPIC_MOISTURE, msg, true);

  Serial.print("[SOIL] moisture=");
  Serial.println(moisture);
}

void setup() {
  pinMode(PUMP_PIN, OUTPUT);
  digitalWrite(PUMP_PIN, LOW);

  Serial.begin(9600);

  randomSeed(analogRead(A0));

  Wire.begin();
  if (soil.begin()) Serial.println("[SOIL] Sensor detected");
  else Serial.println("[SOIL] Not detected (check Qwiic cable / I2C)");

  connectWifiOnce();
  connectMqttOnce();

  publishMoistureNow();
  lastPublishMs = millis();
}

void loop() {
  connectWifiOnce();
  connectMqttOnce();

  client.loop();

  unsigned long now = millis();

  if (digitalRead(PUMP_PIN) == HIGH) {
    if (!pumpOn) pumpOn = true;
    if (pumpStartedAtMs == 0) pumpStartedAtMs = now;
  }

  if (pumpOn) {
    if ((long)(now - pumpOffAtMs) >= 0 || (long)(now - pumpStartedAtMs) >= (long)PUMP_MAX_MS) {
      pumpStopNow();
    }
  } else {
    digitalWrite(PUMP_PIN, LOW);
  }

  if (now - lastPublishMs >= PUBLISH_INTERVAL_MS) {
    publishMoistureNow();
    lastPublishMs = now;
  }

  delay(20);
}
