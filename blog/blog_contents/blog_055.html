#include "arduino_secrets.h"

char ssid[] = SECRET_SSID;
char pass[] = SECRET_PASS;
int status = WL_IDLE_STATUS;

char hueHubIP[] = "172.22.151.226"; // colunm 12
String hueUserName = "...";

WiFiClient wifi;
HttpClient httpClient = HttpClient(wifi, hueHubIP);

int lastSentBrightness = -1;
int lastSentHue = -1;
bool lightOn = true;

int lastButtonState = LOW;

void setup() {
  Serial.begin(9600);
  while (!Serial);

  pinMode(2, INPUT);

  while (status != WL_CONNECTED) {
    Serial.print("Connecting to: ");
    Serial.println(ssid);
    status = WiFi.begin(ssid, pass);
  }
  Serial.println("Connected.");
}

void loop() {
  // 按钮检测
  int buttonState = digitalRead(2);
  if (buttonState == HIGH && lastButtonState == LOW) {
    lightOn = !lightOn;
    Serial.print("Button pressed | light -> ");
    Serial.println(lightOn ? "ON" : "OFF");
    sendOnOff(4, lightOn);
  }
  lastButtonState = buttonState;

  // 亮度和色相（只在灯开着时才读取发送）
  if (lightOn) {
    int rawBri = analogRead(A0);
    int brightness = map(rawBri, 0, 1023, 0, 254);

    int rawHue = analogRead(A1);
    long hueVal = map(rawHue, 0, 1023, 0, 65535);

    Serial.print("Bri raw: "); Serial.print(rawBri);
    Serial.print(" | bri: "); Serial.print(brightness);
    Serial.print(" | Hue raw: "); Serial.print(rawHue);
    Serial.print(" | hue: "); Serial.print(hueVal);

    bool shouldPut = false;

    if (lastSentBrightness == -1 || abs(brightness - lastSentBrightness) > 10) {
      Serial.print(" | bri changed");
      shouldPut = true;
    }
    if (lastSentHue == -1 || abs(hueVal - lastSentHue) > 2500) {
      Serial.print(" | hue changed");
      shouldPut = true;
    }

    if (shouldPut) {
      Serial.print(" | PUTTING");
      sendState(4, brightness, hueVal);
      lastSentBrightness = brightness;
      lastSentHue = hueVal;
    }

    Serial.println();
  }

  delay(200);
}

void sendOnOff(int light, bool on) {
  String request = "/api/" + hueUserName;
  request += "/lights/";
  request += light;
  request += "/state/";

  String contentType = "application/json";
  String hueCmd = "{\"on\":" + String(on ? "true" : "false") + "}";

  httpClient.put(request, contentType, hueCmd);

  int statusCode = httpClient.responseStatusCode();
  String response = httpClient.responseBody();

  Serial.print("Status: "); Serial.print(statusCode);
  Serial.print(" | Response: "); Serial.println(response);
}

void sendState(int light, int brightness, long hueVal) {
  String request = "/api/" + hueUserName;
  request += "/lights/";
  request += light;
  request += "/state/";

  String contentType = "application/json";
  String hueCmd = "{\"bri\":" + String(brightness) + ",\"hue\":" + String(hueVal) + "}";

  httpClient.put(request, contentType, hueCmd);

  int statusCode = httpClient.responseStatusCode();
  String response = httpClient.responseBody();

  Serial.print(" | Status: "); Serial.print(statusCode);
  Serial.print(" | Response: "); Serial.println(response);
}
