// ── POST to VM ──────────────────────────────────────────────
bool postToVM(String date, float lunar, String name) {
  WiFiSSLClient client;
  client.setTimeout(10000);

  if (!client.connect(VM_SERVER, VM_PORT)) {
    Serial.println("[VM] Connection failed");
    return false;
  }

  String body = "{\"date\":\"" + date + "\",\"lunar\":" + String(lunar, 4) + ",\"name\":\"" + name + "\"}";  // manual joint of JSON body

  // HTTP/1.0 reqeust
  client.println("POST /log HTTP/1.0");
  client.print("Host: ");
  client.println(VM_SERVER);
  client.println("Content-Type: application/json");
  client.print("Content-Length: ");
  client.println(body.length());
  client.println("Connection: close");
  client.println();
  client.print(body);

  // waiting for response
  unsigned long timeout = millis();
  while (client.connected() || client.available()) {
    if (millis() - timeout > 10000) break;
    if (client.available()) {
      String line = client.readStringUntil('\n');
      Serial.println("[VM] " + line);
      break;
    }
  }

  client.stop();
  Serial.println("[VM] Posted: " + body);
  return true;
}
